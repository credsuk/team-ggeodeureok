async function runDynamicIssueAnalysis(query) {
    const container = document.getElementById("rawSearchResultsContainer");
    const countEl = document.getElementById("rawSearchResultCount");
    const timeEl = document.getElementById("rawSearchDbMs");

    if (!container) {
        console.error("?? ?? ??? ??? ??? ?? ?????.");
        return;
    }

    container.innerHTML = `
        <div class="rounded-xl border border-sky-200 bg-sky-50 p-5">
            <p class="font-bold text-sky-950">?? ?? ?? ?? ?</p>
            <p class="mt-2 text-sm text-sky-900">
                ?? ??? ???? ??????? ??? ???? ????.
                ??? ??? ?? ? ? ?? ?? ? ????.
            </p>
            <p class="mt-2 text-xs text-sky-800">???: ${escapeHtml(query)}</p>
        </div>`;
    if (countEl) countEl.textContent = "?? ?";
    if (timeEl) timeEl.textContent = "...";

    const started = Date.now();

    try {
        const response = await fetch("/api/analyze", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ query }),
        });

        const data = await response.json();
        if (!response.ok) {
            throw new Error(data.detail || `HTTP ${response.status}`);
        }

        const issue = data.issue || {};
        const summary = data.summary || {};
        const forecast = data.forecast || {};
        const quality = data.quality || {};
        const propagation = Array.isArray(data.propagation) ? data.propagation : [];
        const monthly = Array.isArray(data.monthly_series) ? data.monthly_series : [];
        const evidence = Array.isArray(data.evidence) ? data.evidence : [];

        const safe = (value) => escapeHtml(String(value ?? "-"));
        const number = (value) => {
            const n = Number(value);
            return Number.isFinite(n) ? n.toLocaleString("ko-KR") : "-";
        };

        const maxMonthly = Math.max(
            1,
            ...monthly.map(item => Number(
                item.mention_count ?? item.count ?? item.mentions ?? item.value ?? 0
            ) || 0)
        );

        const monthlyHtml = monthly.length
            ? monthly.map(item => {
                const label = item.month ?? item.month_start ?? item.period ?? item.date ?? "-";
                const value = Number(
                    item.mention_count ?? item.count ?? item.mentions ?? item.value ?? 0
                ) || 0;
                const width = Math.max(0, Math.min(100, value / maxMonthly * 100));
                return `
                    <div class="grid grid-cols-[85px_1fr_48px] items-center gap-2 text-xs">
                        <span class="text-slate-700">${safe(label)}</span>
                        <div class="h-3 overflow-hidden rounded bg-slate-100">
                            <div class="h-full rounded bg-sky-600" style="width:${width}%"></div>
                        </div>
                        <span class="text-right font-mono font-bold">${number(value)}</span>
                    </div>`;
            }).join("")
            : '<p class="text-sm text-slate-600">?? ?? ???? ????.</p>';

        const propagationHtml = propagation.length
            ? propagation.map((item, index) => `
                <div class="rounded-lg border border-slate-200 bg-white p-3">
                    <div class="flex flex-wrap items-center justify-between gap-2">
                        <span class="font-bold text-slate-950">
                            ${index + 1}. ${safe(item.council)}
                        </span>
                        <span class="text-xs text-slate-600">
                            ${safe(item.stage)}
                        </span>
                    </div>
                    <div class="mt-1 text-sm text-slate-700">
                        관측 기간: ${safe(item.first_observation_date)}
                        ~ ${safe(item.last_observation_date)}
                        · 언급 건수: ${number(item.mention_count)}
                    </div>
                </div>`).join("")
            : '<p class="text-sm text-slate-600">??? ?? ???? ????.</p>';

        const evidenceItems = evidence.slice(0, 5);
        const evidenceHtml = evidenceItems.length
            ? evidenceItems.map((item, index) => {
                const text = item.text ?? item.excerpt ?? item.preview ?? "";
                const council = item.council ?? item.assembly ?? "-";
                const date = item.meeting_date ?? item.date ?? "-";
                const meeting = item.meeting_name ?? item.meetingName ?? "";
                const agenda = item.agenda_title ?? "";
                return `
                    <article class="rounded-lg border border-slate-200 bg-white p-3">
                        <div class="font-bold text-slate-900">
                            ${index + 1}. ${safe(council)}
                            <span class="font-normal text-slate-600">${safe(date)}</span>
                        </div>
                        <p class="mt-1 text-xs text-slate-600">${safe(meeting)}</p>
                        <p class="mt-1 text-xs text-slate-600">안건: ${safe(agenda || "-")}</p>
                        <p class="mt-2 whitespace-pre-wrap text-sm leading-relaxed text-slate-800">${safe(text || "?? ??? ????.")}</p>
                    </article>`;
            }).join("")
            : '<p class="text-sm text-slate-600">?? ?? ??? ????. ??? ?? ??? ?????.</p>';

        const interval = Array.isArray(forecast.prediction_interval_80)
            ? forecast.prediction_interval_80
            : [];

        container.innerHTML = `
            <section class="space-y-4">
                <div class="rounded-xl border border-amber-300 bg-amber-50 p-4">
                    <div class="flex flex-wrap items-center justify-between gap-2">
                        <h3 class="text-lg font-black text-slate-950">${safe(issue.issue_name || query)}</h3>
                        <span class="rounded-full bg-amber-200 px-3 py-1 text-xs font-bold text-amber-950">
                            ?? ? ?? ??
                        </span>
                    </div>
                    <p class="mt-2 text-sm text-slate-800">
                        ? ??? ?? ?? ????, ?? ???? ????? ????? ???? ????.
                    </p>
                </div>

                <div class="grid grid-cols-2 gap-3 md:grid-cols-4">
                    <div class="rounded-xl border bg-white p-3">
                        <p class="text-xs text-slate-600">?? ??</p>
                        <p class="mt-1 text-2xl font-black">${number(summary.council_count)}</p>
                    </div>
                    <div class="rounded-xl border bg-white p-3">
                        <p class="text-xs text-slate-600">?? ??</p>
                        <p class="mt-1 text-2xl font-black">${number(summary.evidence_unit_count)}</p>
                    </div>
                    <div class="rounded-xl border bg-white p-3">
                        <p class="text-xs text-slate-600">?? ??</p>
                        <p class="mt-1 text-sm font-bold">${safe(summary.first_observation_date)}</p>
                    </div>
                    <div class="rounded-xl border bg-white p-3">
                        <p class="text-xs text-slate-600">??? ??</p>
                        <p class="mt-1 text-sm font-bold">${safe(summary.last_observation_date)}</p>
                    </div>
                </div>

                <div class="rounded-xl border border-slate-200 bg-slate-50 p-4">
                    <h4 class="mb-3 font-black text-slate-950">??? ?? ?? ??</h4>
                    <div class="space-y-2">${propagationHtml}</div>
                    <p class="mt-3 text-xs text-slate-600">
                        ?? ??? ????, ? ??? ?? ??? ??? ???? ??? ????.
                    </p>
                </div>

                <div class="rounded-xl border border-slate-200 bg-white p-4">
                    <h4 class="mb-3 font-black text-slate-950">?? ?? ??</h4>
                    <div class="space-y-2">${monthlyHtml}</div>
                </div>

                <div class="rounded-xl border border-slate-200 bg-white p-4">
                    <h4 class="font-black text-slate-950">?? ??? ??</h4>
                    <div class="mt-2 text-sm text-slate-800">
                        ?? ?: <b>${safe(forecast.target_month)}</b>
                        ? ??: <b>${safe(forecast.direction)}</b>
                    </div>
                    <p class="mt-2 text-2xl font-black text-sky-800">
                        ${number(forecast.predicted_mention_count)}?
                    </p>
                    <p class="text-sm text-slate-700">
                        80% ?? ??: ${interval.length >= 2
                            ? `${number(interval[0])}~${number(interval[1])}?`
                            : "???? ??"}
                    </p>
                    <p class="mt-2 text-xs text-amber-800">
                        ${safe(forecast.warning || "?? ??? ?? ?? ??????.")}
                    </p>
                </div>

                <div class="rounded-xl border border-slate-200 bg-white p-4">
                    <h4 class="mb-3 font-black text-slate-950">?? ?? ??</h4>
                    <div class="grid grid-cols-2 gap-2 text-sm">
                        <p>?? ????: <b>${safe(quality.seed_coverage)}</b></p>
                        <p>?? ??? ??: <b>${safe(quality.cluster_similarity)}</b></p>
                        <p>?? ?? ???: <b>${number(quality.valid_unique_member_count)}</b></p>
                        <p>?? ??: <b>${number(quality.invalid_date_count)}</b></p>
                    </div>
                </div>

                <div class="rounded-xl border border-slate-200 bg-white p-4">
                    <h4 class="mb-3 font-black text-slate-950">?? ??</h4>
                    <div class="space-y-3">${evidenceHtml}</div>
                </div>

                <div class="flex flex-wrap items-center justify-between gap-3">
                    <p class="text-xs text-slate-600">
                        ?? ??: ${number(data.elapsed_ms ?? (Date.now() - started))} ms
                    </p>
                    <button id="dynamicFallbackSearchBtn" type="button"
                        class="rounded-lg border border-slate-300 bg-white px-4 py-2 text-sm font-bold text-slate-800 hover:bg-slate-50">
                        ?? ?? ???? ??
                    </button>
                </div>
            </section>`;

        if (countEl) countEl.textContent = `${number(summary.council_count)}? ??`;
        if (timeEl) timeEl.textContent = number(data.elapsed_ms ?? (Date.now() - started));

        document.getElementById("dynamicFallbackSearchBtn")?.addEventListener("click", () => {
            executeRawVectorSearch();
        });

    } catch (error) {
        console.error("?? ?? ?? ??:", error);
        container.innerHTML = `
            <div class="rounded-xl border border-red-300 bg-red-50 p-4">
                <h3 class="font-black text-red-900">?? ??? ??????.</h3>
                <p class="mt-2 whitespace-pre-wrap text-sm text-red-800">${escapeHtml(error.message || "?? ??")}</p>
                <button id="dynamicRetryBtn" type="button"
                    class="mt-3 rounded-lg border border-red-300 bg-white px-4 py-2 text-sm font-bold text-red-900">
                    ?? ??
                </button>
                <button id="dynamicRawSearchBtn" type="button"
                    class="ml-2 mt-3 rounded-lg border border-slate-300 bg-white px-4 py-2 text-sm font-bold text-slate-800">
                    ?? ?? ??
                </button>
            </div>`;

        if (countEl) countEl.textContent = "??";
        if (timeEl) timeEl.textContent = "-";

        document.getElementById("dynamicRetryBtn")?.addEventListener("click", () => {
            runDynamicIssueAnalysis(query);
        });
        document.getElementById("dynamicRawSearchBtn")?.addEventListener("click", () => {
            executeRawVectorSearch();
        });
    }
}
